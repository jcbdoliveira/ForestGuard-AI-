def get_risk_color(level: str, prob: float) -> str:
    """v4 — escala do mapa polígono: Baixo(verde) Médio(amarelo) Alto(laranja) Maior(vermelho)."""
    if prob >= 0.75 or level == "CRITICO":
        return "#dc2626"  # Maior
    if prob >= 0.50 or level == "ALTO":
        return "#f97316"  # Alto
    if prob >= 0.25 or level == "MODERADO":
        return "#eab308"  # Médio
    return "#22c55e"  # Baixo (verde)

        avail_h = ch - margin_top - margin_bottom
        # células quadradas que cabem inteiras na tela
        side = min(avail_w / n_c, avail_h / n_r)

        grid_h = side * n_r

        origin_y = margin_top + (avail_h - grid_h) / 2
        gap = max(1.0, side * 0.04)

        font_size = max(6, min(11, int(side * 0.22)))
        font_id = max(5, min(8, int(side * 0.14)))

        for c in inside:

            prob = sc.get("probability", 0.0)
            level = sc.get("attention_level", "BAIXO")
            col = c.get("col", 0)
            row = c.get("row", 0)
            x1 = origin_x + (col - min_c) * side + gap / 2
            y1 = origin_y + (max_r - row) * side + gap / 2
            x2 = x1 + side - gap
            y2 = y1 + side - gap

            tag_name = f"cell_{c['cell_id']}"

                x1, y1, x2, y2, fill=color, outline=outline_col, width=outline_w, tags=(tag_name, "cell")

    def _cell_google_maps_url(cell: dict) -> str:
        """
        Abre o Google Maps com o retângulo da célula desenhado (rota fechada azul)
        usando os 4 cantos do bbox da grade.
        bbox da célula: [min_lat, min_lon, max_lat, max_lon]
        """
        lat, lon = cell["centroid"]
        bb = cell.get("bbox")
        if bb and len(bb) >= 4:
            min_lat, min_lon, max_lat, max_lon = float(bb[0]), float(bb[1]), float(bb[2]), float(bb[3])
            # 4 cantos + fecha o polígono (Google desenha a linha em azul)
            corners = [
                (min_lat, min_lon),
                (min_lat, max_lon),
                (max_lat, max_lon),
                (max_lat, min_lon),
                (min_lat, min_lon),
            ]
            path = "/".join(f"{a:.6f},{b:.6f}" for a, b in corners)
            # zoom ~17 para enquadrar uma célula ~1km
            return (
                f"https://www.google.com/maps/dir/{path}"
                f"/@{lat:.6f},{lon:.6f},17z/data=!3m1!4b1!4m2!4m1!3e2"
            )
        return f"https://www.google.com/maps/@{lat},{lon},17z"

def abrir_mapa_satelite_embutido(cell: dict):
    """
    Abre uma nova janela pop-up interna com o mapa de satélite do Google
    e desenha o polígono vermelho (BBox) ao redor da célula degradada.
    """
    # v16 (parecer C3): imports de GUI locais — o módulo continua importável sem display
    import tkinter as tk
    import tkintermapview

    lat, lon = cell["centroid"]
    bb = cell.get("bbox")
    
    # 1. Criar uma nova janela flutuante (Toplevel)
    janela_mapa = tk.Toplevel()
    janela_mapa.title(f"Alerta de Desmatamento - Centro: {lat:.6f}, {lon:.6f}")
    janela_mapa.geometry("800x600")
    janela_mapa.grab_set()  # Foca a atenção do usuário nesta janela
    
    # 2. Criar o widget de mapas dentro da nova janela
    mapa = tkintermapview.TkinterMapView(janela_mapa, width=800, height=600, corner_radius=0)
    mapa.pack(fill="both", expand=True)
    
    # 3. CONFIGURAÇÃO CHAVE: Usar o servidor oficial de Satélite do Google (sem precisar de chaves API!)
    mapa.set_tile_server("https://mt0.google.com/vt/lyrs=s&x={x}&y={y}&z={z}", max_zoom=22)
    
    # 4. Centralizar o mapa no centroide com zoom de 17x (ideal para células de 1km)
    mapa.set_position(lat, lon)
    mapa.set_zoom(17)
    
    # 5. Adicionar um marcador (pin) vermelho no centro
    mapa.set_marker(lat, lon, text="Centro do Alerta")
    
    # 6. Desenhar o quadrado vermelho (BBox)
    if bb and len(bb) >= 4:
        min_lat, min_lon, max_lat, max_lon = float(bb[0]), float(bb[1]), float(bb[2]), float(bb[3])
        
        # Define os 4 cantos do quadrado
        coordenadas_quadrado = [
            (min_lat, min_lon),
            (min_lat, max_lon),
            (max_lat, max_lon),
            (max_lat, min_lon),
            (min_lat, min_lon) # Fecha o polígono voltando ao início
        ]
        
        # Desenha o quadrado vermelho translúcido na tela
        mapa.set_polygon(
            coordenadas_quadrado,
            fill_color=None,            # Mantém o interior transparente para ver a floresta embaixo
            outline_color="#EF4444",    # Borda vermelha vibrante
            border_width=3,
            name="bbox_alerta"
        )
